; alignment = length of the mean heading vector: 1 = all birds fly the same way, ~0 = random headings.
; nn = mean distance from a bird to its nearest neighbour (patches). 300 birds, 600 ticks, 5 runs each.
print "== 1.2 Flocking, one rule switched off at a time"
foreach [ "none (defaults)" "max-separate-turn" "max-align-turn" "max-cohere-turn" "separate and cohere (alignment only)" ] [ v -> let a 0 let nn 0 foreach [1 2 3 4 5] [ s -> set max-separate-turn 1.5 set max-align-turn 5 set max-cohere-turn 3 if v = "max-separate-turn" [ set max-separate-turn 0 ] if v = "max-align-turn" [ set max-align-turn 0 ] if v = "max-cohere-turn" [ set max-cohere-turn 0 ] if v = "separate and cohere (alignment only)" [ set max-separate-turn 0 set max-cohere-turn 0 ] random-seed s setup repeat 600 [ go ] set a a + sqrt ((mean [ sin heading ] of turtles) ^ 2 + (mean [ cos heading ] of turtles) ^ 2) set nn nn + mean [ distance min-one-of other turtles [ distance myself ] ] of turtles ] print (word "set to 0: " v " -> alignment " precision (a / 5) 2 ", nearest-neighbour distance " precision (nn / 5) 2) ]
