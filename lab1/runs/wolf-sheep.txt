; The loops test the model's own stop conditions, so the user-message pop-up is never reached.
print "== 1.1 sheep-wolves, default sliders"
set model-version "sheep-wolves"
foreach [1 2 3 4 5] [ s -> random-seed s setup while [ ticks < 3000 and any? turtles and not (not any? wolves and count sheep > max-sheep) ] [ go ] print (word "seed " s ": ended at tick " ticks ", wolves " count wolves ", sheep " count sheep) ]
print "== 1.1 sheep-wolves-grass, defaults (wolf-reproduce 5, wolf-gain 20, sheep-reproduce 4, sheep-gain 4, grass 30): 1000 ticks"
set model-version "sheep-wolves-grass"
foreach [1 2 3] [ s -> random-seed s setup while [ ticks < 1000 and any? turtles and not (not any? wolves and count sheep > max-sheep) ] [ go ] print (word "seed " s ": tick " ticks ", wolves " count wolves ", sheep " count sheep) ]
let ok 0 foreach range 20 [ s -> random-seed (s + 100) setup while [ ticks < 1000 and any? wolves and any? sheep ] [ go ] if any? wolves and any? sheep [ set ok ok + 1 ] ] print (word "same values, 20 more runs: both alive at tick 1000 in " ok " of 20")
print "== 1.1 sheep-wolves-grass, grass-regrowth-time 100, sheep-gain-from-food 2 (others default)"
set grass-regrowth-time 100 set sheep-gain-from-food 2
foreach [1 2 3 4 5] [ s -> random-seed s setup while [ ticks < 1000 and any? turtles and not (not any? wolves and count sheep > max-sheep) ] [ go ] print (word "seed " s ": tick " ticks ", wolves " count wolves ", sheep " count sheep) ]
print "== 1.1 for contrast: greedy wolves (wolf-reproduce 20, wolf-gain 50), grass defaults"
set grass-regrowth-time 30 set sheep-gain-from-food 4 set wolf-reproduce 20 set wolf-gain-from-food 50
foreach [1 2 3] [ s -> random-seed s setup while [ ticks < 1000 and any? turtles and not (not any? wolves and count sheep > max-sheep) ] [ go ] print (word "seed " s ": tick " ticks ", wolves " count wolves ", sheep " count sheep) ]
